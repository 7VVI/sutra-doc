# KB Knowledge Base Module Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a complete KB (Knowledge Base) management module in RuoYi-Vue-Plus 5.6.0, integrating Elasticsearch full-text search from legacy KM module, with new multi-dimensional permission system and folder hierarchy.

**Architecture:** MySQL stores document metadata with 16 kb_* tables. Elasticsearch (ES 8.x) indexes document content for full-text search with IK analyzer. Permission calculation follows inheritance chain: doc permission > folder permission > parent folder (inherited) > root default > none.

**Tech Stack:** Spring Boot 3.5.12, MyBatis-Plus 3.5.16, Elasticsearch 8.x + IK analyzer, Spring AI Alibaba Document Parser (TikaDocumentParser), MapStruct-Plus 1.5.0, Sa-Token 1.44.0

```java
package com.hmoob.doc.es.config;

import co.elastic.clients.elasticsearch.ElasticsearchClient;
import co.elastic.clients.json.jackson.JacksonJsonpMapper;
import co.elastic.clients.transport.ElasticsearchTransport;
import co.elastic.clients.transport.rest_client.RestClientTransport;
import lombok.RequiredArgsConstructor;
import org.apache.http.HttpHost;
import org.apache.http.auth.AuthScope;
import org.apache.http.auth.UsernamePasswordCredentials;
import org.apache.http.impl.client.BasicCredentialsProvider;
import org.elasticsearch.client.RestClient;
import org.elasticsearch.client.RestClientBuilder;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

/**
 * ES客户端配置
 */
@Configuration
@RequiredArgsConstructor
@ConditionalOnProperty(prefix = "kb.elasticsearch", name = "enabled", havingValue = "true", matchIfMissing = true)
public class ElasticsearchConfig {

    private final ElasticsearchProperties properties;

    @Bean
    public ElasticsearchClient elasticsearchClient() {
        HttpHost httpHost = new HttpHost(properties.getHost(), properties.getPort(), "http");
        RestClientBuilder builder = RestClient.builder(httpHost);

        if (Boolean.TRUE.equals(properties.getAuth())) {
            BasicCredentialsProvider credentialsProvider = new BasicCredentialsProvider();
            credentialsProvider.setCredentials(
                AuthScope.ANY,
                new UsernamePasswordCredentials(properties.getUsername(), properties.getPassword())
            );
            builder.setHttpClientConfigCallback(httpClientBuilder ->
                httpClientBuilder.setDefaultCredentialsProvider(credentialsProvider)
            );
        }

        RestClient restClient = builder.build();
        ElasticsearchTransport transport = new RestClientTransport(
            restClient, new JacksonJsonpMapper()
        );

        return new ElasticsearchClient(transport);
    }
}
```

```java
package com.hmoob.doc.parser.impl;

import com.alibaba.cloud.ai.document.parser.tika.TikaDocumentParser;
import com.hmoob.doc.parser.IKbDocParserService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.ai.document.Document;
import org.springframework.stereotype.Service;

import java.io.InputStream;
import java.util.List;
import java.util.stream.Collectors;

/**
 * 文档解析服务实现
 * 使用 Spring AI Alibaba TikaDocumentParser 解析各种格式的文档
 * 支持: PDF, DOC, DOCX, XLS, XLSX, PPT, PPTX, TXT, HTML 等
 */
@Slf4j
@RequiredArgsConstructor
@Service
public class KbDocParserServiceImpl implements IKbDocParserService {

    private final TikaDocumentParser tikaDocumentParser;

    @Override
    public List<String> parseContent(InputStream inputStream) {
        List<Document> documents = tikaDocumentParser.parse(inputStream);
        return documents.stream()
            .map(Document::getText)
            .collect(Collectors.toList());
    }

    @Override
    public String parseContentAsString(InputStream inputStream) {
        List<Document> documents = tikaDocumentParser.parse(inputStream);
        return documents.stream()
            .map(Document::getText)
            .collect(Collectors.joining("\n"));
    }

    @Override
    public List<Document> parseDocuments(InputStream inputStream) {
        return tikaDocumentParser.parse(inputStream);
    }
}
```

```java
package com.hmoob.doc.controller;

import cn.dev33.satoken.annotation.SaCheckPermission;
import com.hmoob.common.core.domain.R;
import com.hmoob.common.core.validate.AddGroup;
import com.hmoob.common.core.validate.EditGroup;
import com.hmoob.common.log.annotation.Log;
import com.hmoob.common.log.enums.BusinessType;
import com.hmoob.common.mybatis.core.page.PageQuery;
import com.hmoob.common.mybatis.core.page.TableDataInfo;
import com.hmoob.common.web.core.BaseController;
import com.hmoob.doc.domain.bo.KbDocBo;
import com.hmoob.doc.domain.vo.KbDocVo;
import com.hmoob.doc.service.IKbDocService;
import com.hmoob.doc.service.IKbPermissionService;
import jakarta.validation.constraints.NotEmpty;
import lombok.RequiredArgsConstructor;
import org.springframework.validation.annotation.Validated;
import org.springframework.web.bind.annotation.*;

import java.util.List;

/**
 * 文档管理Controller
 */
@Validated
@RequiredArgsConstructor
@RestController
@RequestMapping("/kb/doc")
public class KbDocController extends BaseController {

    private final IKbDocService docService;
    private final IKbPermissionService permissionService;

    /** 查询文档列表 */
    @SaCheckPermission("kb:doc:list")
    @GetMapping("/list")
    public TableDataInfo<KbDocVo> list(KbDocBo bo, PageQuery pageQuery) {
        return docService.selectPageDocList(bo, pageQuery);
    }

    /** 查询目录下的文档 */
    @SaCheckPermission("kb:doc:list")
    @GetMapping("/listByFolder/{folderId}")
    public TableDataInfo<KbDocVo> listByFolder(@PathVariable Long folderId, PageQuery pageQuery) {
        permissionService.checkFolderPermission(folderId, "view");
        return docService.selectPageDocListByFolder(folderId, pageQuery);
    }

    /** 获取文档详情 */
    @SaCheckPermission("kb:doc:query")
    @GetMapping("/{docId}")
    public R<KbDocVo> getInfo(@PathVariable Long docId) {
        permissionService.checkDocPermission(docId, "view");
        return R.ok(docService.selectDocById(docId));
    }

    /** 上传文档 */
    @SaCheckPermission("kb:doc:add")
    @Log(title = "文档管理", businessType = BusinessType.INSERT)
    @PostMapping
    public R<Void> add(@Validated(AddGroup.class) @RequestBody KbDocBo bo) {
        permissionService.checkFolderPermission(bo.getFolderId(), "upload");
        return toAjax(docService.insertDoc(bo));
    }

    /** 修改文档 */
    @SaCheckPermission("kb:doc:edit")
    @Log(title = "文档管理", businessType = BusinessType.UPDATE)
    @PutMapping
    public R<Void> edit(@Validated(EditGroup.class) @RequestBody KbDocBo bo) {
        permissionService.checkDocPermission(bo.getDocId(), "edit");
        return toAjax(docService.updateDoc(bo));
    }

    /** 删除文档 */
    @SaCheckPermission("kb:doc:remove")
    @Log(title = "文档管理", businessType = BusinessType.DELETE)
    @DeleteMapping("/{docIds}")
    public R<Void> remove(@NotEmpty(message = "主键不能为空") @PathVariable Long[] docIds) {
        for (Long docId : docIds) {
            permissionService.checkDocPermission(docId, "delete");
        }
        return toAjax(docService.deleteDocByIds(List.of(docIds)));
    }

    /** 发布文档 */
    @SaCheckPermission("kb:doc:edit")
    @Log(title = "文档管理", businessType = BusinessType.UPDATE)
    @PutMapping("/publish/{docId}")
    public R<Void> publish(@PathVariable Long docId) {
        permissionService.checkDocPermission(docId, "edit");
        return toAjax(docService.publishDoc(docId));
    }

    /** 撤回文档 */
    @SaCheckPermission("kb:doc:edit")
    @Log(title = "文档管理", businessType = BusinessType.UPDATE)
    @PutMapping("/withdraw/{docId}")
    public R<Void> withdraw(@PathVariable Long docId) {
        permissionService.checkDocPermission(docId, "edit");
        return toAjax(docService.withdrawDoc(docId));
    }

    /** 收藏/取消收藏 */
    @PostMapping("/favourite/{docId}")
    public R<Void> toggleFavourite(@PathVariable Long docId) {
        return toAjax(docService.toggleFavourite(docId));
    }
}
```
